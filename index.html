function fazerLoginEmail() { const email = document.getElementById("loginEmail").value.trim(); const senha = document.getElementById("loginSenha").value; const erroEl = document.getElementById("authError"); erroEl.innerText = ""; if (!email || !senha) { erroEl.innerText = "Preencha o e-mail e a senha."; return; } // Tenta fazer o login; se o usuário não existir, cria a conta automaticamente para facilitar o cadastro auth.signInWithEmailAndPassword(email, senha) .catch((error) => { if (error.code === 'auth/user-not-found' || error.code === 'auth/invalid-credential') { return auth.createUserWithEmailAndPassword(email, senha); } else { throw error; } }) .catch((error) => { erroEl.innerText = "Erro: " + error.message; }); } function fazerLoginGoogle() { const provider = new firebase.auth.GoogleAuthProvider(); auth.signInWithPopup(provider).catch((error) => { document.getElementById("authError").innerText = "Erro no Google: " + error.message; }); } function fazerLogout() { auth.signOut().then(() => { // Limpa os campos de login ao sair document.getElementById("loginEmail").value = ""; document.getElementById("loginSenha").value = ""; document.getElementById("authError").innerText = ""; }).catch((error) => { console.error("Erro ao sair:", error); }); }
